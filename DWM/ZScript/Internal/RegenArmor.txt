Class RegenativeArmor : EventHandler
{
	PlayerPawn Player[8];
	int ArmorAmount;
	name ItemArmor[2][4];
	const ITEM_ARMOR_SIZE = 4;
	


	override void WorldLoaded(WorldEvent e)
	{
		//Get the players first
		for (int index = 0; index < MAXPLAYERS; index++)
		{
			if (Players[Index].mo)
			{
				Player[Index] = Players[Index].mo;
			}		
		}
		
		ArmorAmount = 3;
		
		ItemArmor[0][0] = 'RMD_ArmorRegenBase';
		ItemArmor[0][1] = 'RMD_ArmorUpgrade2';
		ItemArmor[0][2] = 'RMD_ArmorUpgrade3';
		ItemArmor[0][3] = 'RMD_ArmorUpgrade4';

		ItemArmor[1][0] = 'RegenArmorCell';
		ItemArmor[1][1] = 'RMD_ArmorCellT2';
		ItemArmor[1][2] = 'RMD_ArmorCellT3';
		ItemArmor[1][3] = 'RMD_ArmorCellT4';
		
	}


	// PLAY scope : collect data
	override void WorldTick()	// PLAY scope
	{
	
		for (int index = 0; index < Player.size(); index++)
		{
			int Tics = 0;

			if (Player[Index])
			{	
				Tics++;
			
				Inventory PArmor = Player[Index].FindInventory("ArmorHitDelay");

				if (Tics >= 2)
				{
					Tics = 0;
				}
				
				if (PArmor)
				{
					int ArmorDelay = PArmor.Amount;
		
					if (ArmorDelay > 0)
					{
						ArmorDelay--;
						Player[Index].SetInventory("ArmorHitDelay", ArmorDelay);
						return;
					}			

					if (ArmorDelay == 0)
					{
						for (int i = 0; i < ITEM_ARMOR_SIZE; i++)
						{
							if (Player[Index].FindInventory(ItemArmor[0][i]))
							{
								Player[Index].A_GiveInventory(ItemArmor[1][i]);			
								
							}
						}					
					}

				}

			}		
		}
	}

}

//The item tracker for armor regeneration
Class ArmorHitDelay : SpecialItem 
{
	Default
	{
		Inventory.Amount 1;
		Inventory.MaxAmount TICRATE * 12;
		+INVENTORY.KEEPDEPLETED //Needed so that the whole system works.		
	}
}
