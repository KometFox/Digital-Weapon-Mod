Class RMD_ArmorBase : ArmorBonus
{
	Default
	{
		Inventory.Pickupmessage "Got some armor?";
		Inventory.PickupSound "Items/SmallArmorCell";
		Armor.Savepercent 50;
		Armor.Saveamount 1;
		Armor.Maxsaveamount 500;
		-COUNTITEM
		-INVENTORY.ALWAYSPICKUP
	}

	override void PostBeginPlay()
	{
		Super.PostBeginPlay();
		
		Angle = Random[Angle](0, 360);	
	}

	
	States
	{
		Spawn:
			IDLE AA 1;
			loop;
	}
}

//The Armor cell for regeneration
Class RMD_ArmorRegenBase : SpecialItem
{
	name CellArmor;
	
	Property CellArmor:CellArmor;

	Default
	{
		Inventory.Pickupmessage "Armor Charge";
		Inventory.PickupSound "Items/SmallArmorCell";
		Inventory.Amount 1;
		Inventory.MaxAmount 1;
		
		RMD_ArmorRegenBase.CellArmor "RMD_ArmorCellT2";
	}
	
	override bool TryPickup(in out Actor Toucher)
	{
		bool received = Super.TryPickup(Toucher);
		if (received)
		{
			Toucher.TakeInventory("BasicArmor", 0xFFFFFF);
			Toucher.A_GiveInventory(CellArmor, 1);
		}
		return received;
	}
}
