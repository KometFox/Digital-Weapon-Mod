class RMD_AVRIL : BaseWeapon
{
	double ScopeZoom;

	Default
	{
		Weapon.SlotNumber 6;
		Weapon.SelectionOrder 1;
		Weapon.AmmoUse 0;
		Weapon.AmmoGive 0;
		Weapon.AmmoType "RMD_Cells";
		Inventory.PickupMessage "$GOTCHAINGUN";
		Inventory.AltHudIcon "GAVRIL";
		Obituary "$OB_MPCHAINGUN";
		Tag "Plasma AVRIL";
		
		BaseWeapon.IronSight IRON_INACTIVE;
		
		BaseWeapon.MagazineMax 5;
		BaseWeapon.Magazine 5;
		
	}

	Action void FireRocket()
	{		
		if (Invoker.IronSight == IRON_INACTIVE)
		{
			A_FireProjectile("RMD_AVRIL_Rocket", 0, 1, 6, 4, CMF_AIMDIRECTION,0);
			B_MuzzleFlash("RMD_GrenadeMuzzleFlash", (10, 3.3, -1.95));	
		}
		else
		{
			A_FireProjectile("RMD_AVRIL_Rocket", 0, 1, 0, 4, CMF_AIMDIRECTION,0);
			B_MuzzleFlash("RMD_GrenadeMuzzleFlash", (7, 0.0, -1.0));	
		}

		Invoker.Magazine -= 1;
		RMD_BARINFO.Set_Mag(invoker.Magazine);
		A_StartSound("AVRIL/Fire", CHAN_WEAPON);
		A_AlertMonsters();	
		B_WpnRecoil(0.6);
	}

	Action void B_InitWeapon()
	{
		RMD_InitWeapon CIWpn = new("RMD_InitWeapon");
		
		CIWpn.AmmoIcon = "AMMOCELL";
		CIWpn.Ammo1 = Invoker.AmmoType1.GetClassName();
		CIWpn.Magazine = Invoker.Magazine;
		Invoker.ScopeZoom = 1.0;
		
		InitWeapon(CIWpn);	
	}
	
	Action void ZoomIn()
	{
		if (Invoker.ScopeZoom <= 8.0)
		{
			A_ZoomFactor(Invoker.ScopeZoom *= 1.1);
			Invoker.ScopeZoom = min(Invoker.ScopeZoom, 8.1);
		}
	}
	
	Action void ZoomOut()
	{
		A_SetCrosshair(0);
		A_ZoomFactor(invoker.ScopeZoom = 1.0, ZOOM_INSTANT);
	}
	
	States
	{

	Init:
		TNT1 A 0 B_InitWeapon();
		Goto ReadyReal;
		
	ReadyReal:
		CHGG A 0;
		IDL1 A 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_NOBOB);
		Loop;	

	Deselect:
		SEL1 HGFEDCBA 1;
		SEL1 A 1 A_Lower(int.max);
		TNT1 A 1 A_Lower(int.max);
		Wait;
		
	Select:
		TNT1 A 0 B_InitWeapon();
		SEL1 ABCDEFGH 1;
		SEL1 H 1 A_Raise(int.max);
		IDL1 A 1 A_Raise(int.max);
        Goto ReadyReal;
			
	//Ironsight Mode
	IronReady:
		SCOA C 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_NOBOB | WRF_ALLOWUSER1);
		Loop;
		
	IronSightDeselect:
		IDL1 A 0 ZoomOut();
		IDL1 A 7;
		Goto ReadyReal;
		
	AltFire:
		TNT1 A 0 B_GoToIronSight("AltHold", "IronSightDeselect");
		Goto Ready;
		
	AltHold:
		SCOA C 1 ZoomIn();
		TNT1 A 0 A_Refire("AltHold");
		Goto IronReady;
		
	Fire:
		TNT1 A 0 FireCheck();
		Goto ReadyReal;
		
	FireNow:
		TNT1 A 0 CheckIfAmmo(Invoker.Magazine, "ReadyReal");
		FIR1 A 1 FireRocket();
		FIR1 ABCDEFGHIJKLM 2;
		FIR1 M 0 A_ReFire;
		Goto ReadyReal;
		
	IronFireNow:
		TNT1 A 0 CheckIfAmmo(Invoker.Magazine, "IronReady");
		SCOA C 1 FireRocket();
		SCOA CCCCCCCCCCCCC 2;
		SCOA C 0 A_ReFire;
		Goto IronReady;
	
	Reload:
		TNT1 A 0 ReloadCheck();
		Goto ReadyReal;
		
	ReloadNow:
		TNT1 A 0 A_TakeInventory("GotoReload", 1);
		REL1 ABCDEFGHI 3;
		TNT1 A 0
		{
			A_StartSound("AVRIL/MagOut", CHAN_WEAPON);
		}
		REL1 JK 2;
		REL1 L 0;
		REL1 L 24;
		REL1 L 0;
		REL1 MN 2;
		TNT1 A 0
		{
			A_StartSound("AVRIL/MagIn", CHAN_WEAPON +1);	
			ReloadGun(Invoker.AmmoType1, 0, 1, 18);
		}
		REL1 OPQRSTUVWXYZ 2;
		Goto ReadyReal;
		

	}
}

Class RMD_AVRIL_Rocket : HeavyExplosivesFast
{
	Default
	{
		Speed 75;
		Decal "RMD_SmallExplosion";

		//Impact 
		BaseProjectileFast.ImpactDamage 0;
		//Splash 
		BaseProjectileFast.SplashDamage 230;
		BaseProjectileFast.FullSplashDamage 60;
		BaseProjectileFast.SplashRadius 180;
		
		BaseProjectileFast.NormalDamageRandom 1.0;
		BaseProjectileFast.NerfDamageRandom 1.0;
		DamageType "Explosive";
		
	}
	
	States
	{
		Spawn:
			IDLE A 0 NoDelay;
			IDLE A 5;
			Goto SpawnLoop;
		SpawnLoop:	
			IDLE A 1 A_SpawnItemEx("RMD_LocustRocketTrail", 0, 0, 0);
			Loop;
		Explode:
			TNT1 A 0 
			{
				A_StartSound("Explosion/Small");
				A_SpawnItemEx("RMD_UTBoomMedBlue", 0, 0, 0);
			}
			Stop;	
	}
}
