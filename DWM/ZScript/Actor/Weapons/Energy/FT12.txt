class RMD_FT12FlameThrower : BaseWeapon
{

	Default
	{
		Weapon.SlotNumber 7;
		Weapon.SelectionOrder 1;
		Weapon.AmmoUse 0;
		Weapon.AmmoGive 0;
		Weapon.AmmoType "RMD_Fuel";
		Inventory.PickupMessage "$GOTCHAINGUN";
		Inventory.AltHudIcon "PIDF12";
		Obituary "$OB_MPCHAINGUN";
		Tag "Dragonfly Flamethrower";
		
		BaseWeapon.IronSight IRON_INACTIVE;
		
		BaseWeapon.MagazineMax 100;
		BaseWeapon.Magazine 100;
		
	}
	
	Action void B_InitWeapon()
	{
		RMD_InitWeapon CIWpn = new("RMD_InitWeapon");
		
		CIWpn.AmmoIcon = "AMMOFUEL";
		CIWpn.Ammo1 = Invoker.AmmoType1.GetClassName();
		CIWpn.Magazine = Invoker.Magazine;

		InitWeapon(CIWpn);	
	}

	Action void FlameGun()
	{
		A_FireProjectile("RMD_FT12_Flame",frandom(-0.2, 0.2), 1, 6, 4, CMF_AIMDIRECTION,frandom(-0.2, 0.2));
			
		//Make it look brighter 
		for (int i = 0; i < 4; i++)
		{
			A_FireProjectile("RMD_FT12_Flame2",frandom(-2.5, 2.5), 1, 6, 4, CMF_AIMDIRECTION,frandom(-2.5, 2.5));
		}
			
		A_AlertMonsters();
	}
	
	Action void ConsumeAmmo()
	{
		Invoker.Magazine -= 1;
		RMD_BARINFO.Set_Mag(invoker.Magazine);		}



	States
	{
	
	Init:
		TNT1 A 0 B_InitWeapon();
		Goto ReadyReal;
		
	ReadyReal:
		CHGG A 0;
		IDLE A 0 A_StopSound(CHAN_WEAPON +1);
		IDLE A 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_NOBOB);
		Loop;	

	Deselect:
		SELC DCBA 1;
		SELC A 1 A_Lower(int.max);
		TNT1 A 0 A_Lower(int.max);
		Wait;
		
	Select:
		TNT1 A 0 B_InitWeapon();
		SELC ABCD 1;
		SELC D 1 A_Raise(int.max);
		IDLE A 1 A_Raise(int.max);
        Goto ReadyReal;
	
	Fire:
		TNT1 A 0 FireCheck();
		Goto ReadyReal;
		
	FireNow:
		IDLE A 0
		{
			A_StartSound("FT12/Start", CHAN_WEAPON);
			A_StartSound("FT12/Loop", CHAN_WEAPON +1, CHANF_LOOP);	
			A_AlertMonsters();
		}
		Goto FireCheck;
		
	FireCheck:
		TNT1 A 0
		{
			if (Invoker.MagLeft(Invoker.Magazine))
			{
				Return A_Jump(256, "FireLoop");
			}
			
			Return A_Jump(256, "ReadyReal");
		}
		Goto ReadyReal;
	
	FireLoop:
		TNT1 A 0 ConsumeAmmo();
		FIRE AA 2 FlameGun();
		FIRE A 0 A_ReFire("FireCheck");
		FIRE BC 2;
		FIRE D 2;
		TNT1 A 0 A_StopSound(CHAN_WEAPON +1);
		TNT1 A 0 A_StartSound("FT12/Stop", CHAN_WEAPON);
		Goto ReadyReal;
		
	
	Reload:
		TNT1 A 0 ReloadCheck();
		Goto ReadyReal;

	ReloadNow:
		TNT1 A 0 A_TakeInventory("GotoReload", 1);
		REL1 AB 3;
		TNT1 A 0
		{
			A_StartSound("FT12/MagOut", CHAN_WEAPON);
		}
		REL1 CDEFGHIJ 3;
		TNT1 A 0
		{
			A_StartSound("FT12/MagIn", CHAN_WEAPON +2);
			
			ReloadGun(Invoker.AmmoType1, 0, 1, 1);
		}
		REL1 K 3;
		Goto ReadyReal;

	}
}


Class RMD_FT12_Flame : Actor
{

		int Timer;
		double FireScale;
		
		Property Timer:Timer;

Default
	{
		Height 6;
		Radius 6;
		Scale 0.08;
		Speed 27;
		Alpha 1.0;
		PROJECTILE;
		+MISSILE;
		+BRIGHT;
		+BOUNCEONACTORS;
		+ROLLSPRITE;
		+FORCEXYBILLBOARD;
		+NOEXTREMEDEATH;
		+NOBLOOD;
		+NOBLOODDECALS;
		+BLOODLESSIMPACT;
		-BLOODSPLATTER;
		-RIPPER;
		+THRUACTORS;
		+DONTSPLASH;
		Damage 0;
		DamageType "Fire";
		DeathType "Fire";
		RenderStyle "Add";
		
		BounceType "Hexen";
		BounceCount 50;
		BounceFactor 0.42;
		WallBounceFactor 0.42;
		
		ProjectileKickBack 0;
	}
	
	Override Void BeginPlay()
	{
		Super.BeginPlay();
		A_SetTranslation("FireOrange");
	}

	void FireAttack()
	{
		int GiveRadius = 85;	
		
		A_Explode(1 + FRandom[FX](0, 5), 80, XF_NOSPLASH);
		
		BlockThingsIterator it = BlockThingsIterator.Create(self, GiveRadius);
		while (it.Next())
		{
			let obj = it.thing;
				
			if ((obj.bISMONSTER) && (!target || obj != target) && obj.health > 0 && Distance3D(obj) <= GiveRadius)
			{
				RMD_FireInv.AttachFire(Obj, Self, "RMD_FireDeath", "Burning");
			}
		}
	}	
	
	
	override void Tick()
	{
		if (Level.IsFrozen() == true)
		{	
			return;
		}
		

		float Scaly = 1 + (1.77 / 10);
		
		Scale.X *= Scaly;
		Scale.Y *= Scaly;	
	
		Super.Tick();
	}

	States
	{
		Spawn:
			TNT1 AA 0;
		Goto Loopy;
		Loopy:
			XBM9 ABCDEFGHIJKLMNOPQRSTUV 2 
			{
				FireAttack();	

				A_FadeOut(1 / 18, 0);			
			}
			Goto Gone; 
		Death:
			TNT1 A 0;
			Stop;
		Gone:
			TNT1 A 0;
			Stop;
	}
}
