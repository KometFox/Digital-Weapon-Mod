class RMD_ASMDShockRifle : BaseWeapon
{
	double ScopeZoom;
	const AmmoConsume = 2; 
	const AltAmmoMag = 4;
	const ImpDamage = 189;
	BEAMZ_LaserBeam beam;


	enum E_FireMode
	{
		Beanz,
		Ball,	
	}

	Default
	{
		Weapon.SlotNumber 7;
		Weapon.SelectionOrder 1;
		Weapon.AmmoUse 0;
		Weapon.AmmoGive 0;
		Weapon.AmmoType1 "RMD_Cells";
		Weapon.AmmoType2 "RMD_Cells";
		Inventory.PickupMessage "$GOTCHAINGUN";
		Inventory.AltHudIcon "PASMD";
		Obituary "$OB_MPCHAINGUN";
		Tag "ASMD Shock Rifle";
		
		BaseWeapon.IronSight IRON_INACTIVE;
		BaseWeapon.FireMode Beanz;
		
		BaseWeapon.MagazineMax 40;
		BaseWeapon.Magazine 40;
		+ALWAYSPUFF;	
		
		DamageType "Plasma";
		DeathType "Plasma";		
	}
	
	//Blam blam
	Action Void FireBullets()
	{
	
	
		Invoker.Magazine -= 1;
		RMD_BARINFO.Set_Mag(invoker.Magazine);
		
		A_FireBullets(0.5, 0.5, -1, ImpDamage, "RMD_BulletPlasmaImpact", FBF_NoRandom); 
		
		
			
		if (Invoker.IronSight == IRON_INACTIVE)
		{				
			// Create Laser
			invoker.beam = invoker.beam.BeamAdd(Self, 5, 2.0, -0.75, 0, 2, type:"ASMDLaser");
			invoker.beam.trackPSprite = false;
			invoker.beam.aimWithWeapon = false; 
			invoker.beam.SetEnabled(true);

			B_MuzzleFlash("RMD_PlasmaFlash", (10, 2.0, -0.7));	
		}
		else if (Invoker.IronSight == IRON_ACTIVE)
		{
			// Create Laser
			invoker.beam = invoker.beam.BeamAdd(Self, 5, 0.0, -0.75, 0, 2, type:"ASMDLaser");
			invoker.beam.trackPSprite = false;
			invoker.beam.aimWithWeapon = false; 
			invoker.beam.SetEnabled(true);
		}
		A_AlertMonsters();
		A_StartSound("ASMD/Fire", CHAN_WEAPON);
	}
	
	Action void FirePlasma()
	{
		if (Invoker.Magazine <= AltAmmoMag)
		{
			return;
		}
	
		if (Invoker.IronSight == IRON_INACTIVE)
		{
			A_FireProjectile("ASMD_MiniPlasmaBall", 0, 1, 6, 4, CMF_AIMDIRECTION, 0);
			B_MuzzleFlash("RMD_HeavyPlasmaFlash", (15, 6.0, -2.0));	
		}
		else if (Invoker.IronSight == IRON_ACTIVE)
		{
			A_FireProjectile("ASMD_MiniPlasmaBall", 0, 1, 0, 4, CMF_AIMDIRECTION, 0);
		}
		
		Invoker.Magazine -= AltAmmoMag;
		RMD_BARINFO.Set_Mag(invoker.Magazine);	
		A_StartSound("ASMD/AltFire", CHAN_WEAPON);
		A_AlertMonsters();
	}	
	
	
	Action void B_InitWeapon()
	{
		RMD_InitWeapon CIWpn = new("RMD_InitWeapon");
		
		CIWpn.AmmoIcon = "AMMOCELL";
		CIWpn.Ammo1 = Invoker.AmmoType1.GetClassName();
		CIWpn.Magazine = Invoker.Magazine;
		invoker.ScopeZoom = 1.0;
		
		InitWeapon(CIWpn);	
	}
	
	Action void ZoomIn()
	{
		if ( Invoker.ScopeZoom <= 8.0 )
		{
			A_ZoomFactor(Invoker.ScopeZoom *= 1.1);
			Invoker.ScopeZoom = min(Invoker.ScopeZoom, 8.1);
		}
	}
	
	Action void ZoomOut()
	{
		A_SetCrosshair(0);
		A_ZoomFactor(invoker.ScopeZoom = 1.0, ZOOM_INSTANT);
	}
	
	Action void SwitchFireMode()
	{
		if (Invoker.FireMode == Ball)
		{
			Invoker.FireMode = Beanz;
			RMD_BARINFO.SetFireMode("Shock Beam");
			
		}
		else
		{
			Invoker.FireMode = Ball;
			RMD_BARINFO.SetFireMode("Shock Ball");
		}
		
	}

	Action State FireCheck()
	{	
		if (Invoker.FireMode == Beanz)
		{
			
			if (Invoker.IronSight == IRON_ACTIVE)
			{
				Return ResolveState("IronFireNow");
			}
			else
			{
				Return ResolveState("FireNow");				
			}
				
			Return ResolveState("ReadyReal");
		}

		else if (Invoker.FireMode == Ball)
		{
			
			if (Invoker.IronSight == IRON_ACTIVE)
			{
				Return ResolveState("AltIronFireNow");
			}
			else
			{
				Return ResolveState("AltFireNow");				
			}
				
			Return ResolveState("ReadyReal");
		}
			
		Return ResolveState("ReadyReal");		
	}


	States
	{

	Init:
		IDLE A 0 B_InitWeapon();
		Goto ReadyReal;
	
	ReadyReal:
		IDLE A 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_NOBOB | WRF_ALLOWUSER1);
		Loop;
		
	//Ironsight Mode
	IronReady:
		SCOA D 1 A_WeaponReady(WRF_NOBOB | WRF_ALLOWUSER1);
		Loop;
		
	Deselect:
		SEL1 EDCBA 1 ;
		SEL1 A 1 A_Lower(int.max);
		TNT1 A 0 A_Lower(int.max);
		Wait;
		
	Select:
		TNT1 A 0 B_InitWeapon();
		SEL1 ABCDE 1;
		SEL1 A 1 A_Raise(int.max);
		IDLE A 1 A_Raise(int.max);
        Goto ReadyReal;
			
	//Ironsight Mode
	IronSightDeselect:
		IDLE A 0 ZoomOut();
		IDLE A 7;
		Goto ReadyReal;
		
	AltFire:
		TNT1 A 0 B_GoToIronSight("AltHold", "IronSightDeselect");
		Goto ReadyReal;
		
	AltHold:
		SCOA D 1 ZoomIn();
		TNT1 A 0 A_Refire("AltHold");
		Goto IronReady;
	
	Fire:
		TNT1 A 0 FireCheck();
		Goto ReadyReal;

	User1: //Select Firemode
		"####" "#" 0 SwitchFireMode();
		"####" "#" 6;
		"####" "#" 0;
		Goto Ready;

	//Bullets
	FireNow:
		TNT1 A 0 CheckIfAmmo(Invoker.Magazine, "ReadyReal");
		FIR1 A 1 FireBullets();
		FIR1 BCDEFG 1;
		FIR1 A 17;
		FIR1 A 0 A_ReFire("FireNow");
		Goto ReadyReal;
	
	IronFireNow:
		TNT1 A 0 CheckIfAmmo(Invoker.Magazine, "IronReady");
		SCOA D 1 FireBullets();
		SCOA DDDDD 1;
		SCOA D 17;
		SCOA D 0 A_ReFire("IronFireNow");
		Goto IronReady;	


	AltFireNow:
		TNT1 A 0 CheckIfAmmo(Invoker.Magazine, "ReadyReal");
		FIR1 A 1 FirePlasma();
		FIR1 BCDEFG 1;
		FIR1 A 17;
		FIR1 A 0 A_ReFire("AltFireNow");
		Goto ReadyReal;
	
	AltIronFireNow:
		TNT1 A 0 CheckIfAmmo(Invoker.Magazine, "IronReady");
		SCOA D 1 FirePlasma();
		SCOA DDDDD 1;
		SCOA D 17;
		SCOA D 0 A_ReFire("AltIronFireNow");
		Goto IronReady;	
		
	Reload:
		TNT1 A 0 ReloadCheck();
		Goto ReadyReal;
		
	//The Magazine
	ReloadNow:
		TNT1 A 0 A_TakeInventory("GotoReload", 99);
		REL1 ABCDEFG 2;
		TNT1 A 0
		{
			A_StartSound("M16A3/MagOut", CHAN_WEAPON +3);
			A_StartSound("E12PR/CellOut", CHAN_WEAPON +1);
			A_FireProjectile("EmptyMagSpawn",-5,0,8,-4);
		}
		REL1 HIJKLNNOPQRS 2;
		TNT1 A 0
		{
			A_StartSound("M16A3/MagIn", CHAN_WEAPON +4);
			A_StartSound("E12PR/CellIn", CHAN_WEAPON +2);
			
			ReloadGun(Invoker.AmmoType1, 0, 1, AmmoConsume);
		}
		REL1 TUVWXYZ 2;
		REL2 AB 2;
		Goto ReadyReal;

	
	//FuckingMother
	Template:
		IRND A 0;
		IFIN A 0;
		FIB1 A 0;
		Goto Ready;
	}
}


Class ASMD_MiniPlasmaBall : LightExplosivesFast
{
	const FLARE = SXF_ABSOLUTEVELOCITY + SXF_ABSOLUTEANGLE + SXF_TRANSFERPITCH;

	Default
	{
		Speed 90;
		Scale 1;
		Decal "RMD_SmallExplosion";
		MissileType "RMD_BlueRocketTrail";
		
		//Impact 
		BaseProjectileFast.ImpactDamage 0;
		//Splash 
		BaseProjectileFast.SplashDamage 280;
		BaseProjectileFast.FullSplashDamage 120;
		BaseProjectileFast.SplashRadius 180;
		//Randomizer
		BaseProjectileFast.NerfDamageRandom 1.0;
		BaseProjectileFast.NormalDamageRandom 1.0;
		BaseProjectileFast.HurtPlayer false; 
		
		DamageType "Plasma";
		DeathType "Plasma";
	}
	
	States
	{
		Spawn:
			TNT1 A 0 NoDelay;
			Goto SpawnLoop;
		SpawnLoop:	
			IDLE A 1;
			Loop;
		Explode:
			TNT1 A 0 
			{
				A_StartSound("Plasma/SmallExplosion");
				A_SpawnItemEx("RMD_UTBoomMedBlue", 0, 0, 0);
			}

			Stop;	
	}
}
