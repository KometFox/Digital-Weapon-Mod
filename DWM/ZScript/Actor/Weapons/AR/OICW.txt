class RMD_XM29OICW : BaseWeapon
{
	Enum E_FireMode
	{
		BULLET = 0, 
		GRENADE = 1,
	}

	Default
	{
		Weapon.SlotNumber 5;
		Weapon.SelectionOrder 1;
		Weapon.AmmoUse 0;
		Weapon.AmmoGive 0;
		Weapon.AmmoType1 "RMD_MediumBullets";
		Weapon.AmmoType2 "RMD_MediumBullets";
		Inventory.PickupMessage "$GOTCHAINGUN";
		Inventory.AltHudIcon "WOICW";
		Obituary "$OB_MPCHAINGUN";
		Tag "XM-29 OICW";
		
		BaseWeapon.IronSight IRON_INACTIVE;
		BaseWeapon.FireMode BULLET;
	
		BaseWeapon.MagazineMax 75;
		BaseWeapon.Magazine 75;	
		
		BaseWeapon.MagazineMax2 10;
		BaseWeapon.Magazine2 10;
	}
	
	action state ReloadCheck()
	{
		if (Invoker.FireMode == GRENADE)
			return B_ReloadGoTo(
								"ReadyReal", 
								"IronReady", 
								"IronSightDeselect", 
								"ReloadNadeNow",
								Invoker.AmmoType2, 
								Invoker.Magazine2,
								Invoker.MagazineMax2, 
								1);
		else 
			return B_ReloadGoTo(
								"ReadyReal", 
								"IronReady", 
								"IronSightDeselect", 
								"ReloadNow",
								Invoker.AmmoType1, 
								Invoker.Magazine,
								Invoker.MagazineMax, 
								1);
	}
	
	
	//Blam blam
	Action Void FireBullets()
	{
		if (Invoker.IronSight == IRON_INACTIVE)
		{
			//A_FireProjectile("RMD_M25_Bullet",frandom(-0.7, 0.7), 1, 6, 4, CMF_AIMDIRECTION,frandom(-0.5, 0.5));
			A_FireBullets(0.7, 0.7, -1, 2*random(6,9), "RMD_BulletImpact", FBF_NoRandom); 
			
			B_MuzzleFlash("RMD_BulletMuzzleFlash", (13, 2.0, -2.0));	
			C_SpawnCasing("RifleCase", (12, 6.5, -2.0));
		}
		
		else if (Invoker.IronSight == IRON_ACTIVE)
		{
			//A_FireProjectile("RMD_M25_Bullet",frandom(-0.3, 0.3), 1, 0, 4, CMF_AIMDIRECTION,frandom(-0.3, 0.3));
			A_FireBullets(0.3, 0.3, -1, 2*random(6,9), "RMD_BulletImpact", FBF_NoRandom); 
			
			B_MuzzleFlash("RMD_BulletMuzzleFlash",(10, 0, -0.5));
			C_SpawnCasing("RifleCase", (12, -2.5, -2.5));
		}

		Invoker.Magazine -= 1;
		RMD_BARINFO.Set_Mag(invoker.Magazine);	
		A_StartSound("OICW/Fire", CHAN_WEAPON, 0, 0.85, 2.5);
		A_AlertMonsters();
	}

	//Ka-Pow
	Action Void FireGrenade()
	{
		if (Invoker.IronSight == IRON_INACTIVE)
		{
			A_FireProjectile("RMD_OICW_Grenade",frandom(-0.6, 0.6), 1, 0, 4, CMF_AIMDIRECTION,frandom(-0.6, 0.6));

			B_MuzzleFlash("RMD_GrenadeMuzzleFlash", (15, 3.0, -2.0));	
			//B_SpawnCasing("GrenadeCase", (12, -7.0, -2.0));
		}
		else if (Invoker.IronSight == IRON_ACTIVE)
		{
			A_FireProjectile("RMD_OICW_Grenade",frandom(-0.2, 0.2), 1, 0, 4, CMF_AIMDIRECTION,frandom(-0.2, 0.2));	

			B_MuzzleFlash("RMD_GrenadeMuzzleFlash", (15, 0, -0.5));	
			//B_SpawnCasing("GrenadeCase", (12, -4.0, -0.5));
		}
		
		Invoker.Magazine2 -= 1;
		RMD_BARINFO.Set_Mag(invoker.Magazine2);		
		A_StartSound("OICW/FireGrenade", CHAN_WEAPON);
	}
	
	
	
	Action Void SetAmmoIcon(int Type)
	{
		Switch(Type)
		{
			case BULLET:
				RMD_BARINFO.Set_Mag(invoker.Magazine);
				RMD_BARINFO.SetAmmoIcon("AMMOAR");
				RMD_BARINFO.SetAmmo1("RMD_MediumBullets");
				Break;
				
			case GRENADE:
				RMD_BARINFO.Set_Mag(invoker.Magazine2);
				RMD_BARINFO.SetAmmoIcon("AMMONADE");
				RMD_BARINFO.SetAmmo1("RMD_MediumBullets");
				Break;
		}
	
	
	}
	
	Action void SwitchFireMode()
	{
		if (Invoker.FireMode == BULLET)
		{
			Invoker.FireMode = GRENADE;
			SetAmmoIcon(GRENADE);
		}
		else
		{
			Invoker.FireMode = BULLET;
			SetAmmoIcon(BULLET);
		}
	}
	
	Action void B_InitWeapon()
	{
		RMD_InitWeapon CIWpn = new("RMD_InitWeapon");
		
		CIWpn.AmmoIcon = "AMMOPIST";
		CIWpn.Ammo1 = Invoker.AmmoType1.GetClassName();
		CIWpn.Magazine = Invoker.Magazine;
		
		InitWeapon(CIWpn);
		SetAmmoIcon(Invoker.FireMode);
	}
	
	Action State FireCheck()
	{	
		if (Invoker.FireMode == BULLET)
		{
			
			if (Invoker.IronSight == IRON_ACTIVE)
			{
				Return ResolveState("IronFireNow");
			}
			else
			{
				Return ResolveState("FireNow");				
			}
				
			Return ResolveState("ReadyReal");
		}

		else if (Invoker.FireMode == GRENADE)
		{	
			if (Invoker.IronSight == IRON_ACTIVE)
			{
				Return ResolveState("IronFireGrenadeNow");
			}
			else
			{
				Return ResolveState("FireGrenadeNow");				
			}
				
			Return ResolveState("ReadyReal");
		}
			
		Return ResolveState("ReadyReal");		
	}


	States
	{

	Init:
		IDLE A 0 B_InitWeapon();
		Goto ReadyReal;
	
	ReadyReal:
		IDLE A 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_NOBOB | WRF_ALLOWUSER1);
		Loop;
		
	//Ironsight Mode
	IronReady:
		SCOA D 1 A_WeaponReady(WRF_NOBOB | WRF_ALLOWUSER1);
		Loop;
		
	Deselect:
		SEL1 FEDCBA 1;
		SEL1 A 1 A_Lower(int.max); 
		Wait;
		
	Select:
		TNT1 A 0 B_InitWeapon();
		SEL1 A 1;
		SEL1 ABCDEF 1 A_Raise(int.max);
		SEL1 F 1;
        Goto ReadyReal;
			
	User1: //Select Firemode
		"####" "#" 0 SwitchFireMode();
		"####" "#" 6;
		"####" "#" 0;
		Goto Ready;

	//Ironsight Mode
	IronSightAnim:
		SCOA D 0 A_SetCrosshair(99);
		SCOA DDDD 2 A_ZoomFactor(2.25);
		Goto IronReady;
		
	IronSightDeselect:
		SCOA D 0  A_SetCrosshair(0);
		SCOA DDDD 2 A_ZoomFactor(1.0);
		Goto ReadyReal;
		
	AltFire:
		TNT1 A 0 B_GoToIronSight("IronSightAnim", "IronSightDeselect");
		Goto ReadyReal;
		
	Fire:
		TNT1 A 0 FireCheck();
		Goto ReadyReal;

	//Bullets
	FireNow:
		TNT1 A 0 CheckIfAmmo(Invoker.Magazine, "ReadyReal");
		FIR1 A 1 FireBullets();
		FIR1 BCD 1;
		FIR1 D 0 A_ReFire("FireNow");
		FIR1 EF 1;
		Goto ReadyReal;
		
	IronFireNow:
		TNT1 A 0 CheckIfAmmo(Invoker.Magazine, "IronReady");
		SCOA D 1 FireBullets();
		SCOA DDD 1;
		SOCA D 0 A_ReFire("IronFireNow");
		SCOA DD 1;
		Goto IronReady;	
		
	//Grenade
	FireGrenadeNow:
		TNT1 A 0 CheckIfAmmo(Invoker.Magazine2, "ReadyReal");
		FIG1 A 1 FireGrenade();
		FIG1 BCDEFGHIJKL 2;
		FIG1 L 0 A_ReFire("FireGrenadeNow");
		Goto ReadyReal;
		
	IronFireGrenadeNow:
		TNT1 A 0 CheckIfAmmo(Invoker.Magazine2, "IronReady");
		SCOA D 1 FireGrenade();
		SCOA DDDDDDDDDDD 2;
		SCOA D 0 A_ReFire("IronFireGrenadeNow");
		Goto IronReady;		
		
		
	Reload:
	TNT1 A 0 ReloadCheck();			
	Goto ReadyReal;
		
	//The Magazine
	ReloadNow:
		REL1 ABCDE 2;
		TNT1 A 0
		{
			A_StartSound("OICW/MagOut", CHAN_WEAPON +1);
			A_FireProjectile("EmptyMagSpawn",-5,0,8,-4);
		}
		REL1 HIJKLM 2;
		TNT1 A 0
		{
			A_StartSound("OICW/MagIn", CHAN_WEAPON +2);
			ReloadGun(Invoker.AmmoType1, 0, 1, 1);
		}
		REL1 NOPQ 2;		
		Goto ReadyReal;


	//The Grenade Drum
	ReloadNadeNow:
		REG1 ABCDEF 2;
		TNT1 A 0
		{
			A_StartSound("OICW/MagOut", CHAN_WEAPON);
		}
		REG1 GHIJKL 2;
		TNT1 A 0
		{
			A_StartSound("OICW/MagIn", CHAN_WEAPON +1);
			
			ReloadGun(Invoker.AmmoType2, 1, 1, 5);
		}
		REG1 MNOPQR 2;		
		Goto ReadyReal;		
		
	//FuckingMother
	Template:
		IRND A 0;
		IFIN A 0;
		Goto Ready;
	}
}




Class RMD_OICW_Grenade : LightExplosivesFast
{
	Default
	{
		Speed 85;
		Decal "RMD_SmallExplosion";
		Gravity 0.35;
		//Impact 
		BaseProjectileFast.ImpactDamage 0;
		//Splash 
		BaseProjectileFast.SplashDamage 100;
		BaseProjectileFast.FullSplashDamage 50;
		BaseProjectileFast.SplashRadius 80;
		//Randomizer
		BaseProjectileFast.NormalDamageRandom 1.4;
		BaseProjectileFast.NerfDamageRandom 1.4;
		DamageType "Explosive";
	}
	
	States
	{
		Spawn:
			TNT1 A 0 NoDelay;
			Goto SpawnLoop;
		SpawnLoop:	
			IDLE A 1 A_SpawnItemEx("RMD_25mmTrail", 0, 0, 0);
			Loop;
		Explode:
			TNT1 A 0 
			{
				A_StartSound("Explosion/Small", 5);
				A_SpawnItemEx("RMD_UTBoomSmall", 0, 0, 0);
			}
			Stop;	
	}
}
