Class RMD_RocketTrail : RMD_BaseParticle
{
	
	States
	{
		Spawn:
			TNT1 A 0 NoDelay;
			TNT1 A 0 
			{

				for (int i = 0; i < 4; i++)
				{
					Vector3 PVel = (FRandom[FX](-0.1, 0.1), FRandom[FX](-0.1, 0.1), FRandom[FX](-0.1, 0.1)).unit()*FRandom[FX](0, 0);
					Vector3 RPos = (frandom(0, 0), frandom(0, 0), frandom(0, 0));
					RMD_BaseParticle Particle = RMD_BaseParticle(Spawn("RMD_RocketTrail_Fire", Pos + RPos));
					
					Particle.Vel = PVel;
					Particle.Scale = (0.5, 0.5);
					Particle.ScaleGrowth = 0.92;
					Particle.SetShade("ff3300");
				}
				
				
			}
			Stop;
	}


}

Class RMD_RocketTrail_Fire : RMD_ModelSparks
{
	Default
	{
		RenderStyle "AddShaded";
		+Bright;
		Gravity -0.2;
		Alpha 1.0;
	}
	
	Override void BeginPlay()
	{
		Super.BeginPlay();
		
		Roll = frandom(0, 360);	
	}	
	
	
	States
	{
		Spawn:
			TNT1 A 0;
			YOM3 AAAAAAA 1
			{
				A_SetScale(Scale.X * 0.94, Scale.Y * 0.94);
				A_FadeOut(0.07);
			}
			Stop;	
	}
}

Class RMD_RocketTrail_Spark : RMD_ModelSparks
{
	Default
	{
		+BRIGHT;
		RenderStyle "AddShaded";
		Alpha 1.0;
		Scale 0.05;
	}

	override void BeginPlay()
	{
		float ScaleV = Frandom[FX](0.05, 0.08);
	
		Scale.X += ScaleV;
		Scale.Y += ScaleV;
		Speed = random(5, 10);
		Gravity = 0;
	}

    States
    {
    Spawn:
	   LEGY A 6 BRIGHT;
       LEGY AAAAA 1 Bright A_FadeOut(0.2);
       stop;
    Death:
       TNT1 A 0;
       Stop;
    }
}
