Class RMD_UTBoomBig : Actor
{
	Mixin Mix_ParticleSpawner;
	
	States
	{
		Spawn:
			TNT1 A 0;
			TNT1 A 0 
			{
				//Explosion				

				for (int i = 0; i < 12; ++i)
				{
					Vector3 PVel = (FRandom[FX](-1.0, 1.0), FRandom[FX](-1.0, 1.0), FRandom[FX](-1.0, 1.0)).unit()*FRandom[FX](0.5, 2.5);
					Vector3 RPos = (frandom(-20, 20), frandom(-20, 20), frandom(-20, 20));
					RMD_BaseParticle Particle = RMD_BaseParticle(Spawn("RMD_UTBoomSprite", RPos + Pos));
					
					Particle.TeleportMove(Pos + RPos, false);
					Particle.Vel = PVel;
					Particle.Roll = Random(0, 360);
					Particle.Scale = (1.08, 1.08);
					Particle.ScaleGrowth = 1.09;
					Particle.A_SetRenderStyle(1.0, STYLE_AddShaded);
					Particle.SetShade("ff7416");
				}
				
				let CVARS = RMD_CVARHandler(StaticEventHandler.Find('RMD_CVARHandler'));
				int CVAR_LowQuality = CVARS.RMDV_ParticleLowQuality;;
		
				if (CVAR_LowQuality == 1)
				{
					Return;
				}
				
				for (int i = 0; i < 8; ++i)
				{
					Vector3 PVel = (FRandom[FX](-1.0, 1.0), FRandom[FX](-1.0, 1.0), FRandom[FX](-0.3, 0.3)).unit()*FRandom[FX](4.0, 9.0);
					Vector3 RPos = (frandom(-12, 12), frandom(-12, 12), frandom(-12, 12));
					RMD_BaseParticle Particle = RMD_BaseParticle(Spawn("RMD_DoomSmoke", Pos + RPos));
										
					Particle.Vel = PVel;
					Particle.Scale = (1.4, 1.4);
					Particle.TTL = 22;
					Particle.ScaleGrowth = 1.02;
					Particle.A_SetRenderStyle(1.0, Style_Shaded);
					Particle.SetShade("7c7c7c");
				}	
				
				for (int i = 0; i < 25; ++i)
				{
					Actor Part;
					
					Part = A_SpawnProjectile("RMD_TSpark", 2, 0, frandom(0,1) * frandom (0, 360), 2, frandom(0,1)*frandom (30, 360));
				
					if (Part)
					{
						Part.Speed = 35;					
					}
				}
							
				
			}
			Stop;
			
	}
}
