Class RMD_UTBoomSmallBlue : Actor
{
	Mixin Mix_ParticleSpawner;
	
	States
	{
		Spawn:
			TNT1 A 0;
			TNT1 A 0 
			{
				//Explosion				
				for (int i = 0; i < 8; ++i)
				{
					Vector3 PVel = (FRandom[FX](-1.0, 1.0), FRandom[FX](-1.0, 1.0), FRandom[FX](-1.0, 1.0)).unit()*FRandom[FX](0.5, 2.5);
					Vector3 RPos = (frandom(-12, 12), frandom(-12, 12), frandom(-12, 12));
					RMD_BaseParticle Particle = RMD_BaseParticle(Spawn("RMD_NeoBoom", RPos + Pos));
					
					Particle.TeleportMove(Pos + RPos, false);
					Particle.Vel = PVel;
					Particle.Roll = Random(0, 360);
					Particle.Scale = (0.75, 0.75);
					Particle.ScaleGrowth = 1.00;
					Particle.A_SetRenderStyle(1.0, Style_AddShaded);
					Particle.SetShade("268aff");
				}
				
				let CVARS = RMD_CVARHandler(StaticEventHandler.Find('RMD_CVARHandler'));
				int CVAR_LowQuality = CVARS.RMDV_ParticleLowQuality;;
		
				if (CVAR_LowQuality == 1)
				{
					Return;
				}
				
				for (int i = 0; i < 8; ++i)
				{
					Vector3 PVel = (FRandom[FX](-1.0, 1.0), FRandom[FX](-1.0, 1.0), FRandom[FX](-0.3, 0.3)).unit()*FRandom[FX](2.5, 6.0);
					Vector3 RPos = (frandom(-0, 0), frandom(-0, 0), frandom(-0, 0));
					RMD_BaseParticle Particle = RMD_BaseParticle(Spawn("RMD_DoomSmoke", Pos + RPos));
										
					Particle.Vel = PVel;
					Particle.Scale = (0.35, 0.35);
					Particle.TTL = 22;
					Particle.ScaleGrowth = 1.02;
					Particle.A_SetRenderStyle(1.0, Style_Shaded);
					Particle.SetShade("fffdfc");
				}	
				
				//Sparks
				for (int i = 0; i < 12; ++i)
				{
					A_SpawnProjectile("RMD_TSpark", 2, 0, frandom(0,1) * frandom (0, 360), 2, frandom(0,1)*frandom (30, 360));
				
				}
				for (int i = 0; i < 20; ++i)
				{
					A_SpawnProjectile("RMD_ShinySpark", 2, 0, frandom(0,1) * frandom (0, 360), 2, frandom(0,1)*frandom (30, 360));
				
				}			
				
			}
			Stop;
			
	}
}
