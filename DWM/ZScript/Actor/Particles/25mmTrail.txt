Class RMD_25mmTrail : RMD_BaseParticle
{
	
	States
	{
		Spawn:
			TNT1 A 0 NoDelay;
			TNT1 A 0 
			{

				for (int i = 0; i < 2; i++)
				{
					Vector3 PVel = (FRandom[FX](-0.1, 0.1), FRandom[FX](-0.1, 0.1), FRandom[FX](-0.1, 0.1)).unit()*FRandom[FX](0, 0);
					Vector3 RPos = (frandom(0, 0), frandom(0, 0), frandom(0, 0));
					RMD_BaseParticle Particle = RMD_BaseParticle(Spawn("RMD_RocketTrail_Fire", Pos + RPos));
					
					Particle.Vel = PVel;
					Particle.Scale = (0.17, 0.17);
					Particle.ScaleGrowth = 0.92;
					Particle.SetShade("ff3300");
				}
				
				for (int i = 0; i < 2; ++i)
				{
					Vector3 PVel = (FRandom[FX](-1.0, 1.0), FRandom[FX](-1.0, 1.0), FRandom[FX](-1.0, 1.0)).unit()*FRandom[FX](0.0, 0.75);
					Vector3 RPos = (frandom(-0, 0), frandom(-0, 0), frandom(-0, 0));
					RMD_BaseParticle Particle = RMD_BaseParticle(Spawn("RMD_DoomSmoke", Pos + RPos));
										
					Particle.Vel = PVel;
					Particle.Scale = (0.12, 0.12);
					Particle.TTL = 8;
					Particle.ScaleGrowth = 1.08;
					Particle.A_SetRenderStyle(1.0, Style_Shaded);
					Particle.SetShade("fffdfc");
				}				
				
				
			}
			Stop;
	}


}
